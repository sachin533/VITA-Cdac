import java.util.Arrays;

public class Qu1{

    public static int[] getUniq(int arr[]){
        int n = arr.length-1;
        int temp[] = new int[n];
        int head = 0;
        for(int i=0; i<n; i++){
            Boolean isduplicate = false;
            for(int j=0; j<head; j++){
                if(arr[i]== temp[head]){
                    isduplicate = true;
                    break;
                }
            }
             if(!isduplicate){
                    temp[head++]=arr[i];
                }
        }
        return Arrays.copyOf(temp, head);
    }
    public static void main(String args[]){
        int arr[] = { 3, 5, 3, 7, 9, 7, 3, 1 };
        int uniq[] = getUniq(arr);

        System.out.print(Arrays.toString(uniq));

    }
}

public class Qu1 {

    public static void getAns(int arr[]) {
        int start = 0;
        int end = arr.length - 1;
        int temp[] = new int[arr.length];
        int k = 0;

        while (start < end) {
            if (arr[start] < arr[end]) {
                temp[k++] = arr[end];
                temp[k++] = arr[start];
            } else {
                temp[k++] = arr[start];
                temp[k++] = arr[end];
            }
            start++;
            end--;
        }

       
        if (start == end) {
            temp[k] = arr[start];
        }

        for (int i = 0; i < temp.length; i++) {
            System.out.print(temp[i] + " ");
        }
    }

    public static void main(String args[]) {
        int arr[] = {1, 2, 3, 4, 5, 6, 7};
        getAns(arr);
    }
}
